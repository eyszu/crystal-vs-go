Language benchmark · 2026-11-11

Crystal vs Go, in 20 tests

Both languages compile to native code. The same 20 programs were written in each, run on one machine, and checked to produce identical results before any timing counted.

The short version

1.35× Crystal faster
Geometric mean of Go time ÷ Crystal time across all 20 tests. Above 1.0 means Crystal finished sooner.

9 Crystal wins   5 Go wins   6 ties

A win means more than 10% faster. Anything closer counts as a tie, because repeat runs on this shared VM vary by 5 to 15%.

Crystal's biggest leads

  • Regular expressions 13.4× faster in Crystal (63.0 ms vs 846 ms)
  • Channel ping-pong 4.6× faster in Crystal (29.3 ms vs 136 ms)
  • JSON parse (typed) 2.5× faster in Crystal (74.7 ms vs 188 ms)

Go's biggest leads

  • N-body simulation 1.6× faster in Go (294 ms vs 189 ms)
  • JSON serialize 1.5× faster in Go (49.2 ms vs 32.2 ms)
  • Hash map (string keys) 1.3× faster in Go (98.2 ms vs 77.1 ms)

Plain number crunching (loops, math, recursion, dynamic programming) is mostly a dead heat. The largest gaps appear in library-heavy work: regular expressions, JSON, string handling, memory allocation and task switching. Those depend on which libraries each language ships, not on how well the compiler optimizes a loop.

What the groups tell you

Compute

Both compilers turn simple loops into very similar machine code, so five of the eight tests land within 10%. Crystal checks integer overflow on every + and * and still keeps pace. Go is ahead on the float-heavy N-body simulation and the sieve. Crystal is ahead on recursive Fibonacci.

Data structures

The binary-trees test allocates millions of small objects. Crystal finished 1.4× sooner than Go, and Go got slower still when limited to one thread, most likely because its garbage collector normally uses the second core. Hash maps split by key type: Crystal led with integer keys, Go led with string keys. Go sorted 2 million integers about 10% faster.

Strings, regex, JSON

The regex gap is the largest result here. Crystal calls PCRE2 with just-in-time compilation. Go's built-in engine guarantees linear-time matching, which costs speed on these patterns. Crystal also parses JSON into typed objects faster. Go serializes faster.

Concurrency

Crystal fibers switch cheaply on one thread. Go goroutines can run in parallel, which pays off for CPU-bound work that this test set does not include. For message passing and spawning tasks, Crystal was faster here.

Read these before quoting the numbers

Crystal is an older version

Ubuntu's package provides Crystal 1.11.2, from April 2024. Go is 1.24.7. A current Crystal release may do better.

Idiomatic code, not hand-tuned

Each program uses normal standard-library calls in its language. Neither side uses unsafe indexing or custom allocators. Experts could squeeze either one further.

One small machine

A shared 2-core cloud VM. Treat gaps under 10% as noise and re-run on your own hardware before deciding anything.

Not covered

Compile time, binary size, startup time, multi-core scaling, network and disk I/O, and developer productivity. Crystal's release build of this program took about 15 seconds. Build time was not measured for Go.

One surprise from the work itself: in Crystal, 3 * n with an Int64 n is computed as a 32-bit integer and overflows, so the Collatz test had to be written n * 3. It crashed with an overflow error until fixed. That is also why every test compares a checksum across languages before any timing counts.

func fib(n int) int {
	if n < 2 {
		return n
	}
	return fib(n-1) + fib(n-2)
}

func benchFib() int64 { return int64(fib(38)) }

func benchCollatz() int64 {
	const limit = 2_000_000
	var bestI, bestLen int64
	for i := int64(1); i < limit; i++ {
		n := i
		steps := int64(0)
		for n != 1 {
			if n&1 == 0 {
				n >>= 1
			} else {
				n = 3*n + 1
			}
			steps++
		}
		if steps > bestLen {
			bestLen = steps
			bestI = i
		}
	}
	return bestI*1000 + bestLen
}

func benchSieve() int64 {
	const n = 20_000_000
	sieve := make([]bool, n+1)
	for i := range sieve {
		sieve[i] = true
	}
	sieve[0], sieve[1] = false, false
	for i := 2; i*i <= n; i++ {
		if sieve[i] {
			for j := i * i; j <= n; j += i {
				sieve[j] = false
			}
		}
	}
	var count int64
	for _, p := range sieve {
		if p {
			count++
		}
	}
	return count
}

func benchMandelbrot() int64 {
	const size = 1000
	const maxIter = 100
	var inside int64
	for py := 0; py < size; py++ {
		ci := 2.0*float64(py)/size - 1.0
		for px := 0; px < size; px++ {
			cr := 3.0*float64(px)/size - 2.0
			zr, zi := 0.0, 0.0
			i := 0
			for ; i < maxIter && zr*zr+zi*zi <= 4.0; i++ {
				zr, zi = zr*zr-zi*zi+cr, 2.0*zr*zi+ci
			}
			if i == maxIter {
				inside++
			}
		}
	}
	return inside
}

func advance(dt float64, x, y, z, vx, vy, vz, m []float64) {
	n := len(x)
	for i := 0; i < n; i++ {
		for j := i + 1; j < n; j++ {
			dx := x[i] - x[j]
			dy := y[i] - y[j]
			dz := z[i] - z[j]
			d2 := dx*dx + dy*dy + dz*dz
			mag := dt / (d2 * math.Sqrt(d2))
			vx[i] -= dx * m[j] * mag
			vy[i] -= dy * m[j] * mag
			vz[i] -= dz * m[j] * mag
			vx[j] += dx * m[i] * mag
			vy[j] += dy * m[i] * mag
			vz[j] += dz * m[i] * mag
		}
	}
	for i := 0; i < n; i++ {
		x[i] += dt * vx[i]
		y[i] += dt * vy[i]
		z[i] += dt * vz[i]
	}
}

func energy(x, y, z, vx, vy, vz, m []float64) float64 {
	e := 0.0
	n := len(x)
	for i := 0; i < n; i++ {
		e += 0.5 * m[i] * (vx[i]*vx[i] + vy[i]*vy[i] + vz[i]*vz[i])
		for j := i + 1; j < n; j++ {
			dx := x[i] - x[j]
			dy := y[i] - y[j]
			dz := z[i] - z[j]
			e -= m[i] * m[j] / math.Sqrt(dx*dx+dy*dy+dz*dz)
		}
	}
	return e
}

func benchNbody() int64 {
	const pi = 3.141592653589793
	const solarMass = 4 * pi * pi
	const dpy = 365.24
	x := []float64{0, 4.84143144246472090e+00, 8.34336671824457987e+00, 1.28943695621391310e+01, 1.53796971148509165e+01}
	y := []float64{0, -1.16032004402742839e+00, 4.12479856412430479e+00, -1.51111514016986312e+01, -2.59193146099879641e+01}
	z := []float64{0, -1.03622044471123109e-01, -4.03523417114321381e-01, -2.23307578892655734e-01, 1.79258772950371181e-01}
	vx := []float64{0, 1.66007664274403694e-03 * dpy, -2.76742510726862411e-03 * dpy, 2.96460137564761618e-03 * dpy, 2.68067772490389322e-03 * dpy}
	vy := []float64{0, 7.69901118419740425e-03 * dpy, 4.99852801234917238e-03 * dpy, 2.37847173959480950e-03 * dpy, 1.62824170038242295e-03 * dpy}
	vz := []float64{0, -6.90460016972063023e-05 * dpy, 2.30417297573763929e-05 * dpy, -2.96589568540237556e-05 * dpy, -9.51592254519715870e-05 * dpy}
	m := []float64{solarMass, 9.54791938424326609e-04 * solarMass, 2.85885980666130812e-04 * solarMass, 4.36624404335156298e-05 * solarMass, 5.15138902046611451e-05 * solarMass}

	px, py, pz := 0.0, 0.0, 0.0
	for i := range m {
		px += vx[i] * m[i]
		py += vy[i] * m[i]
		pz += vz[i] * m[i]
	}
	vx[0] = -px / solarMass
	vy[0] = -py / solarMass
	vz[0] = -pz / solarMass

	for s := 0; s < 3_000_000; s++ {
		advance(0.01, x, y, z, vx, vy, vz, m)
	}
	return int64(math.Round(energy(x, y, z, vx, vy, vz, m) * 1e9))
}

func benchMatmul() int64 {
	const n = 400
	a := make([]float64, n*n)
	b := make([]float64, n*n)
	c := make([]float64, n*n)
	for i := 0; i < n; i++ {
		for j := 0; j < n; j++ {
			a[i*n+j] = float64((i + j) % 10)
			b[i*n+j] = float64((i * j) % 10)
		}
	}
	for i := 0; i < n; i++ {
		for j := 0; j < n; j++ {
			s := 0.0
			for k := 0; k < n; k++ {
				s += a[i*n+k] * b[k*n+j]
			}
			c[i*n+j] = s
		}
	}
	total := 0.0
	for _, v := range c {
		total += v
	}
	return int64(total)
}

func queens(n int, all, cols, d1, d2 int64) int64 {
	if cols == all {
		return 1
	}
	var count int64
	avail := all &^ (cols | d1 | d2)
	for avail != 0 {
		bit := avail & -avail
		avail ^= bit
		count += queens(n, all, cols|bit, ((d1|bit)<<1)&all, (d2|bit)>>1)
	}
	return count
}

func benchNqueens() int64 {
	const n = 14
	return queens(n, (int64(1)<<n)-1, 0, 0, 0)
}

func benchLCS() int64 {
	const n = 8000
	a := make([]byte, n)
	b := make([]byte, n)
	x := uint64(12345)
	for i := range a {
		x = x*lcgA + lcgC
		a[i] = 'A' + byte((x>>33)&3)
	}
	for i := range b {
		x = x*lcgA + lcgC
		b[i] = 'A' + byte((x>>33)&3)
	}
	prev := make([]int32, n+1)
	cur := make([]int32, n+1)
	for i := 1; i <= n; i++ {
		ai := a[i-1]
		for j := 1; j <= n; j++ {
			if ai == b[j-1] {
				cur[j] = prev[j-1] + 1
			} else if prev[j] >= cur[j-1] {
				cur[j] = prev[j]
			} else {
				cur[j] = cur[j-1]
			}
		}
		prev, cur = cur, prev
	}
	return int64(prev[n])
}

// ---------------------------------------------------------------- data structures

type Node struct{ l, r *Node }

func bottomUp(d int) *Node {
	if d <= 0 {
		return &Node{}
	}
	return &Node{bottomUp(d - 1), bottomUp(d - 1)}
}

func (n *Node) check() int64 {
	if n.l == nil {
		return 1
	}
	return 1 + n.l.check() + n.r.check()
}

func benchBinaryTrees() int64 {
	const maxDepth = 17
	total := bottomUp(maxDepth + 1).check()
	long := bottomUp(maxDepth)
	for d := 4; d <= maxDepth; d += 4 {
		iters := 1 << (maxDepth - d + 4)
		var c int64
		for i := 0; i < iters; i++ {
			c += bottomUp(d).check()
		}
		total += c
	}
	total += long.check()
	return total
}

	s := a[0] + a[n/2] + a[n-1]

		s += a[i]
	}

}

func benchHashInt() int64 {
	m := make(map[int64]int64)
	x := uint64(7)
	for i := int64(0); i < 1_000_000; i++ {
		x = x*lcgA + lcgC
		m[int64(x>>33)%2_000_000] = i
	}
	var sum int64
	for i := int64(0); i < 2_000_000; i++ {
		if v, ok := m[i]; ok {
			sum += v
		}
	}
	for i := int64(0); i < 2_000_000; i += 2 {
		delete(m, i)
	}
	return sum*7 + int64(len(m))
}

func benchStringBuilder() int64 {
	var sb strings.Builder
	for i := 0; i < 2_000_000; i++ {
		sb.WriteString("item")
		sb.WriteString(strconv.Itoa(i))
		sb.WriteByte(',')
	}
	s := sb.String()
	var count7 int64
	for i := 0; i < len(s); i++ {
		if s[i] == '7' {
			count7++
		}
	}
	var pieces, pieceLen int64
	for _, p := range strings.Split(s, ",") {
		if len(p) > 0 {
			pieces++
			pieceLen += int64(len(p))
		}
	}
	return int64(len(s)) + count7*3 + pieces*7 + pieceLen
}

var logText string
var reEmail = regexp.MustCompile(`[a-z]+[0-9]+@[a-z]+[0-9]+\.example\.com`)
var reStatus = regexp.MustCompile(`status=([0-9]{3})`)

func getLogText() string {
	if logText == "" {
		statuses := [4]int{200, 404, 500, 301}
		var sb strings.Builder
		for i := 0; i < 200_000; i++ {
			st := strconv.Itoa(statuses[i%4])
			if i%10 == 0 {
				sb.WriteString("noise line " + strconv.Itoa(i) + " status=" + st + "\n")
			} else {
				sb.WriteString("user" + strconv.Itoa(i) + "@host" + strconv.Itoa(i%100) + ".example.com visited /path/" + strconv.Itoa(i) + " status=" + st + "\n")
			}
		}
		logText = sb.String()
	}
	return logText
}

func benchRegex() int64 {
	text := getLogText()
	emails := int64(len(reEmail.FindAllStringIndex(text, -1)))
	var statusSum int64
	for _, m := range reStatus.FindAllStringSubmatch(text, -1) {
		v, _ := strconv.Atoi(m[1])
		statusSum += int64(v)
	}
	return emails*1_000_003 + statusSum
}

func getJSON() []byte {
	if jsonBytes == nil {
		var sb strings.Builder
		sb.WriteByte('[')
		for i := 0; i < 50_000; i++ {
			if i > 0 {
				sb.WriteByte(',')
			}
			cents := (i * 37 % 10000) | 1
			active := "true"
			if i%3 == 0 {
				active = "false"
			}
			is := strconv.Itoa(i)
			sb.WriteString(`{"id":` + is + `,"name":"user` + is + `","email":"user` + is + `@example.com","active":` + active)
			sb.WriteString(fmt.Sprintf(`,"score":%d.%02d`, cents/100, cents%100))
			sb.WriteString(`,"tags":["t` + strconv.Itoa(i%7) + `","t` + strconv.Itoa(i%11) + `","t` + strconv.Itoa(i%13) + `"]`)
			sb.WriteString(`,"address":{"city":"City` + strconv.Itoa(i%100) + `","zip":"` + strconv.Itoa(10000+i%90000) + `"}}`)
		}
		sb.WriteByte(']')
		jsonBytes = []byte(sb.String())
	}
	return jsonBytes
}

var shaBuf []byte

func benchSHA256() int64 {
	if shaBuf == nil {
		shaBuf = make([]byte, 32*1024*1024)
		x := uint64(2024)
		for i := range shaBuf {
			x = x*lcgA + lcgC
			shaBuf[i] = byte(x >> 56)
		}
	}
	var acc int64
	for r := 0; r < 4; r++ {
		shaBuf[0] = byte(r)
		d := sha256.Sum256(shaBuf)
		acc = (acc*31 + int64(d[0])*65536 + int64(d[1])*256 + int64(d[2])) % 1_000_000_007
	}
	return acc
}

func benchChanPingPong() int64 {
	a := make(chan int64)
	b := make(chan int64)
	go func() {
		for v := range a {
			b <- v + 1
		}
		close(b)
	}()
	var s int64
	for i := int64(0); i < 300_000; i++ {
		a <- i
		s += <-b
	}
	close(a)
	return s
}

func benchSpawnSum() int64 {
	res := make(chan int64, 1000)

def bench_collatz : Int64
  limit = 2_000_000_i64
  best_i = 0_i64
  best_len = 0_i64
  i = 1_i64
  while i < limit
    n = i
    steps = 0_i64
    while n != 1
      if n & 1 == 0
        n >>= 1
      else
        n = n * 3 + 1
      end
      steps += 1
    end
    if steps > best_len
      best_len = steps
      best_i = i
    end
    i += 1
  end
  best_i * 1000 + best_len
end

def bench_mandelbrot : Int64
  size = 1000
  max_iter = 100
  inside = 0_i64
  size.times do |py|
    ci = 2.0 * py / size - 1.0
    size.times do |px|
      cr = 3.0 * px / size - 2.0
      zr = 0.0
      zi = 0.0
      i = 0
      while i < max_iter && zr * zr + zi * zi <= 4.0
        zr, zi = zr * zr - zi * zi + cr, 2.0 * zr * zi + ci
        i += 1
      end
      inside += 1 if i == max_iter
    end
  end
  inside
end

def advance(dt : Float64, x : Array(Float64), y : Array(Float64), z : Array(Float64),
            vx : Array(Float64), vy : Array(Float64), vz : Array(Float64), m : Array(Float64))
  n = x.size
  i = 0
  while i < n
    j = i + 1
    while j < n
      dx = x[i] - x[j]
      dy = y[i] - y[j]
      dz = z[i] - z[j]
      d2 = dx * dx + dy * dy + dz * dz
      mag = dt / (d2 * Math.sqrt(d2))
      vx[i] -= dx * m[j] * mag
      vy[i] -= dy * m[j] * mag
      vz[i] -= dz * m[j] * mag
      vx[j] += dx * m[i] * mag
      vy[j] += dy * m[i] * mag
      vz[j] += dz * m[i] * mag
      j += 1
    end
    i += 1
  end
  i = 0
  while i < n
    x[i] += dt * vx[i]
    y[i] += dt * vy[i]
    z[i] += dt * vz[i]
    i += 1
  end
end

def energy(x : Array(Float64), y : Array(Float64), z : Array(Float64),
           vx : Array(Float64), vy : Array(Float64), vz : Array(Float64), m : Array(Float64)) : Float64
  e = 0.0
  n = x.size
  i = 0
  while i < n
    e += 0.5 * m[i] * (vx[i] * vx[i] + vy[i] * vy[i] + vz[i] * vz[i])
    j = i + 1
    while j < n
      dx = x[i] - x[j]
      dy = y[i] - y[j]
      dz = z[i] - z[j]
      e -= m[i] * m[j] / Math.sqrt(dx * dx + dy * dy + dz * dz)
      j += 1
    end
    i += 1
  end
  e
end

def queens(n : Int32, all : Int64, cols : Int64, d1 : Int64, d2 : Int64) : Int64
  return 1_i64 if cols == all
  count = 0_i64
  avail = all & ~(cols | d1 | d2)
  while avail != 0
    bit = avail & -avail
    avail ^= bit
    count += queens(n, all, cols | bit, ((d1 | bit) << 1) & all, (d2 | bit) >> 1)
  end
  count
end

end

def bench_lcs : Int64

      if ai == b[j - 1]

      elsif prev[j] >= cur[j - 1]

      else
        cur[j] = cur[j - 1]
      end
    end
    prev, cur = cur, prev
  end
  prev[n].to_i64
end

class Node
  def initialize(@l : Node? = nil, @r : Node? = nil)
  end

  def check : Int64
    if (l = @l) && (r = @r)
      1_i64 + l.check + r.check
    else
      1_i64
    end
  end
end

def bottom_up(d : Int32) : Node
  if d <= 0
    Node.new
  else
    Node.new(bottom_up(d - 1), bottom_up(d - 1))
  end
end

def run_one(cmd, extra_env):
    env = dict(os.environ)
    env.update(extra_env)
    # GNU time launches the benchmark from a tiny process, so its max-RSS figure is the
    # benchmark's own peak (os.wait4 from Python would inherit Python's ~14 MB footprint).
    rss_file = os.path.join(OUT, ".rss")
    t0 = time.monotonic()
    p = subprocess.run(["/usr/bin/time", "-f", "%M", "-o", rss_file] + cmd,
                       stdout=subprocess.PIPE, stderr=subprocess.PIPE, env=env)
    wall = (time.monotonic() - t0) * 1000
    if p.returncode != 0:
        return {"error": p.stderr.decode()[:300] or f"exit status {p.returncode}"}

    out = p.stdout
    rec = json.loads(out.decode().strip().splitlines()[-1])
    t = rec["times_ms"]
    return {
        "median_ms": statistics.median(t),
        "min_ms": min(t),
        "max_ms": max(t),
        "stdev_ms": statistics.pstdev(t),
        "times_ms": t,
        "checksum": rec["checksum"],
        "stable": rec["stable"],
        "peak_rss_mb": rss_kb / 1024.0,
        "process_wall_ms": wall,
    }


def sh(cmd):
    try:
        return subprocess.check_output(cmd, shell=True, text=True, stderr=subprocess.STDOUT).strip()
    except Exception as e:
        return str(e)


def main():
    results = {}
    for bid, cat, title, desc in BENCHES:
        results[bid] = {"category": cat, "title": title, "desc": desc}
        for name, cmd, env in CONFIGS:
            r = run_one(cmd + [bid, str(REPS)], env)
            results[bid][name] = r
            if "error" in r:
                print(f"{bid:15s} {name:8s} ERROR {r['error']}", flush=True)
            else:
                print(f"{bid:15s} {name:8s} median {r['median_ms']:9.1f} ms  rss {r['peak_rss_mb']:7.1f} MB  sd {r['stdev_ms']:.1f}", flush=True)
        a, b = results[bid]["crystal"], results[bid]["go"]
        results[bid]["checksum_match"] = ("error" not in a and "error" not in b and a["checksum"] == b["checksum"] == results[bid]["go_p1"].get("checksum"))
    meta = {
        "date": time.strftime("%Y-%m-%d %H:%M UTC", time.gmtime()),
        "reps": REPS,
        "cpu": sh("lscpu | grep 'Model name' | sed 's/.*: *//'"),
        "cores": os.cpu_count(),
        "mem_mb": int(sh("free -m | awk '/Mem:/{print $2}'") or 0),
        "os": sh("grep PRETTY_NAME /etc/os-release | cut -d'\"' -f2"),
        "kernel": platform.release(),
        "go_version": sh("go version"),
        "crystal_version": sh("crystal --version | head -3 | tr '\\n' ' '"),
    }
    with open(os.path.join(HERE, "results.json"), "w") as f:
        json.dump({"meta": meta, "results": results}, f, indent=1)
    print("wrote results.json")


if __name__ == "__main__":
    main()
